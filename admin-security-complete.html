🔐 JOB CONNECT - Configuration Admin & Modérateur Complète

📋 RÉSUMÉ

Cette guide te montre comment :

  • ✅ Récupérer correctement l'UID de l'utilisateur connecté
  • ✅ Créer un document admin/modérateur dans Firestore
  • ✅ Sécuriser les accès avec les règles Firestore
  • ✅ Vérifier si un utilisateur est admin avant d'afficher le panel

1️⃣ CODE À AJOUTER DANS TON index.html

A) Fonction pour récupérer l'UID actuel

Ajoute ceci dans ta section <script> (après l'initialisation Firebase) :

// ========== UID & ADMIN FUNCTIONS ========== let currentUserUID = null; let isCurrentUserAdmin = false; // Récupérer l'UID de l'utilisateur connecté function getCurrentUID() { return new Promise((resolve) => { const unsubscribe = onAuthStateChanged(auth, (user) => { if (user) { currentUserUID = user.uid; console.log('✅ UID récupéré:', currentUserUID); resolve(currentUserUID); } else { currentUserUID = null; resolve(null); } unsubscribe(); }); }); } // Vérifier si l'utilisateur est admin async function checkIfUserIsAdmin() { if (!currentUserUID) { console.log('❌ Pas d\'utilisateur connecté'); isCurrentUserAdmin = false; return false; } try { const adminDocRef = doc(db, 'admins', currentUserUID); const adminDocSnap = await getDoc(adminDocRef); if (adminDocSnap.exists()) { console.log('✅ Utilisateur est admin'); isCurrentUserAdmin = true; return true; } else { console.log('❌ Utilisateur n\'est pas admin'); isCurrentUserAdmin = false; return false; } } catch (e) { console.error('❌ Erreur lors de la vérification admin:', e); isCurrentUserAdmin = false; return false; } } // Créer un document admin (FONCTION POUR GABRIEL AU DÉMARRAGE) async function createAdminDocument(uid, email, nom) { try { const adminRef = doc(db, 'admins', uid); // Vérifier si le document existe déjà const existingDoc = await getDoc(adminRef); if (existingDoc.exists()) { console.log('⚠️ Ce compte est déjà admin'); return false; } // Créer le document admin await setDoc(adminRef, { uid: uid, email: email, nom: nom, role: 'modérateur', // ou 'admin' dateCreation: serverTimestamp(), permissions: { canViewReports: true, canViewAllAccounts: true, canBanAccounts: true, canManageAdmins: false // Seul l'admin peut changer ça } }); console.log('✅ Document admin créé pour:', uid); isCurrentUserAdmin = true; return true; } catch (e) { console.error('❌ Erreur création admin:', e); return false; } } // Appeler au démarrage pour initialiser async function initializeAdminSystem() { const uid = await getCurrentUID(); if (uid) { await checkIfUserIsAdmin(); // Afficher/cacher le panel admin selon le statut const adminTab = document.querySelector('[data-tab="admin"]'); if (adminTab && isCurrentUserAdmin) { adminTab.style.display = 'block'; } } }

2️⃣ RÈGLES FIRESTORE (SÉCURITÉ ESSENTIELS)

ACCÈS FIREBASE CONSOLE → RÈGLES FIRESTORE

rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // ===== COLLECTION ADMINS ===== match /admins/{uid} { // Lecture: admin peut lire les infos de tous les admins, autres ne voient rien allow read: if request.auth.uid == uid || exists(/databases/$(database)/documents/admins/$(request.auth.uid)); // Écriture: seul l'admin lui-même peut créer son doc (au démarrage) // seul Gabriel (uid hardcodé) peut modifier les permissions allow create: if request.auth.uid == uid && request.resource.data.uid == uid && request.resource.data.email == request.auth.token.email; allow update: if request.auth.uid == "GABRIEL_UID_ICI" || (request.auth.uid == uid && request.resource.data.diff(resource.data).affectedKeys() .hasOnly(['nom'])); // Peut modifier juste son nom allow delete: if request.auth.uid == "GABRIEL_UID_ICI"; } // ===== COLLECTION SIGNALEMENTS ===== match /signalements/{signalementId} { // Lecture: admin peut voir tous, auteur peut voir le sien allow read: if exists(/databases/$(database)/documents/admins/$(request.auth.uid)) || (request.query.filters.length == 1 && request.query.filters[0].field.name == 'telephoneAuteur' && request.query.filters[0].value == request.auth.token.phone_number); // Écriture: tout le monde peut créer un signalement allow create: if request.resource.data.telephoneAuteur == request.auth.token.phone_number && request.resource.data.dossierId != null; // Admin peut mettre à jour le statut allow update: if exists(/databases/$(database)/documents/admins/$(request.auth.uid)); allow delete: if request.auth.uid == "GABRIEL_UID_ICI"; } // ===== AUTRES COLLECTIONS ===== // Garder tes règles existantes pour depot_annonces, utilisateurs, etc. } }

⚠️ IMPORTANT : Remplace GABRIEL_UID_ICI par ton UID réel (voir point 3)

3️⃣ TROUVER TON UID GABRIEL

Étape 1: Se connecter à JOB CONNECT

Connecte-toi avec ton compte (email/mot de passe ou Google)

Étape 2: Ouvrir la console Chrome

Sur Android:
  • Ouvre JOB CONNECT dans Chrome
  • Appuie sur Menu (3 points) → More tools → Inspect
  • Ça peut être difficile sur mobile...

Étape 3: Méthode PLUS FACILE (Firestore Console)

Plutôt que la console Chrome, utilise Firestore directement :

  1. Va sur console.firebase.google.com
  2. Sélectionne ton projet (jobconnect-aade6 ou autre)
  3. Clique sur Authentication (à gauche)
  4. Tu verras tous tes utilisateurs avec leurs UID
  5. Copie l'UID de ton compte

✅ Tu as ton UID!

⚠️ POINTS CRITIQUES

  • NE PUBLIE PAS ton UID réel dans le code côté client — utilise-le que dans les règles Firestore
  • La collection admins doit exister dans Firestore AVANT de tester
  • Les règles Firestore s'appliquent même en développement
  • Ne supprime jamais ton doc admin sans avoir une autre méthode pour te reconnecter
